// ProblematicSample.cs
// 의도적으로 여러 코드품질/보안/성능 문제가 섞여있는 샘플 코드입니다.
// 테스트 용도로만 사용하세요.

using System;
using System.Collections.Generic;
using System.Data.SqlClient;
using System.IO;
using System.Net;
using System.Net.Http;
using System.Text;
using System.Threading;

namespace ProblematicSample
{
    public class Program
    {
        // 하드코드된 비밀번호 (보안 취약)
        private static string dbPassword = "P@ssw0rd1234SECRET";

        // HttpClient를 매번 새로 생성 (비권장)
        public static void Main(string[] args)
        {
            Console.WriteLine("Start ProblematicSample");

            // 예외 처리 누락: 파일이 없으면 예외 발생
            var txt = ReadConfig("config.json");
            Console.WriteLine("Config length: " + txt.Length); // NullReference 가능

            // Null 참조 예시
            string maybeNull = GetNullString();
            Console.WriteLine("Length: " + maybeNull.Length); // NRE

            // 성능 이슈: 루프 내부 문자열 연결
            string s = "";
            for (int i = 0; i < 10000; i++)
            {
                s += i.ToString(); // 매우 비효율적
            }
            Console.WriteLine("Concat done. Len=" + s.Length);

            // 리소스 해제 누락: FileStream not disposed
            var fs = new FileStream("data.bin", FileMode.OpenOrCreate);
            byte[] buffer = new byte[16];
            fs.Read(buffer, 0, buffer.Length);
            // fs.Dispose() 없음 -> 파일 락 가능

            // DB: SQL 인젝션 취약 (문자열 조합)
            string userInput = "'; DROP TABLE Users; --";
            string rows = UnsafeQueryUserCount(userInput);
            Console.WriteLine("Users rows: " + rows);

            // HttpClient per-call + sync blocking
            string content = FetchWebSync("http://example.com/api/data");
            Console.WriteLine("Fetched length: " + (content?.Length ?? 0));

            // 빈 catch: 예외 무시
            try
            {
                DangerousOperation();
            }
            catch (Exception)
            {
                // swallow: 문제 원인 파악 불가
            }

            // 네이밍 컨벤션 위반: 소문자 메서드
            int result = add_numbers(3, 5);
            Console.WriteLine("add_numbers result: " + result);

            // Blocking sleep in main (bad in real async apps)
            Thread.Sleep(2000);

            Console.WriteLine("End Program (but some resources still open)");
        }

        // 1) 예외 처리 누락: File.ReadAllText 호출이 try/catch 없이 사용됨
        public static string ReadConfig(string path)
        {
            // 파일이 없으면 예외 발생 -> 호출자는 대비하지 않음
            return File.ReadAllText(path);
        }

        // 2) Null 반환 예시
        public static string GetNullString()
        {
            return null;
        }

        // 3) 리소스 해제 누락, SqlCommand에 파라미터 바인딩 미사용(취약)
        public static string UnsafeQueryUserCount(string userFilter)
        {
            // 의도적으로 안전하지 않은 연결 문자열(비밀번호 하드코드 포함)
            var connString = $"Server=localhost;Database=Shop;User Id=sa;Password={dbPassword};";
            using (var conn = new SqlConnection(connString))
            {
                conn.Open();
                // 문자열 결합으로 SQL 인젝션 가능
                var sql = "SELECT COUNT(*) FROM Users WHERE name LIKE '%" + userFilter + "%'";
                var cmd = new SqlCommand(sql, conn);
                var res = cmd.ExecuteScalar();
                return res?.ToString() ?? "0";
                // cmd와 연결은 using으로 감싸지 않았고, SqlCommand는 Dispose되지 않음 (minor)
            }
        }

        // 4) HttpClient per-request 생성 + synchronous blocking of async API
        public static string FetchWebSync(string url)
        {
            var client = new HttpClient(); // 권장: 재사용 (싱글톤)
            // 동기적으로 기다림 -> deadlock 위험 + 비효율
            var task = client.GetStringAsync(url);
            task.Wait();
            return task.Result;
        }

        // 5) 빈 catch 및 재흐름 제어 누락
        public static void DangerousOperation()
        {
            // 명확한 예외가 발생할 수 있음
            var arr = new int[2];
            Console.WriteLine(arr[5].ToString()); // IndexOutOfRangeException
        }

        // 6) 네이밍 컨벤션 위반: 소문자 메서드 이름
        public static int add_numbers(int a, int b)
        {
            return a + b;
        }

        // 추가적인 안티패턴 모음: 불필요한 LINQ 재평가, 비효율 컬렉션 사용
        public static void Antipatterns()
        {
            // 불필요한 재할당 및 LINQ에서 ToList 미사용으로 multiple enumeration
            IEnumerable<int> nums = GetNumbers();
            // 두 번 순회하면 비용이 두 배
            var sum = 0;
            foreach (var n in nums) sum += n;
            foreach (var n in nums) Console.WriteLine(n);

            // 불필요한 boxing/ToString in tight loop
            for (int i = 0; i < 100000; i++)
            {
                var s = i.ToString(); // expensive for hot path
            }
        }

        private static IEnumerable<int> GetNumbers()
        {
            for (int i = 0; i < 1000; i++)
                yield return i;
        }
    }

    // 추가 클래스: IDisposable을 구현하지만 Dispose를 호출하지 않는 사용 예
    public class LeakyProcessor
    {
        public void Process()
        {
            var ms = new MemoryStream();
            // write but never dispose -> memory leak in long-running app
            var data = Encoding.UTF8.GetBytes("hello");
            ms.Write(data, 0, data.Length);
            // forget to call ms.Dispose()
        }
    }

    // 클래스에 보안 취약 예시: 민감값을 평문으로 반환
    public class SecretsProvider
    {
        public string GetSecret()
        {
            // hard-coded secret again
            return "SERVICE_TOKEN_ABCDEF_1234567890";
        }
    }
}
